# Workflow V2 · Plan 01a — Động cơ, cầu nối job, compiler, báo giá

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Một run workflow đã publish chạy thật từ đầu đến cuối trên runtime `platform_core/workflow/` — động cơ nhặt wakeup, nộp job cho node app, nhận kết quả qua cầu nối, kết sổ tiền — thay cho thư viện "chỉ lưu" hiện nay.

**Architecture:** Compiler biến draft v3 thành đồ thị `ltt.workflow/3`; báo giá đọc registry; `tao_run` là cửa tạo run duy nhất; động cơ là job `workflow.dong_co` tự nộp lại, xử lý từng run dưới khoá hàng; node job nộp qua `jobs.submit` (không hold riêng) và `jobs.complete/fail/cancel` báo ngược bằng tín hiệu waitpoint + wakeup trong cùng transaction.

**Tech Stack:** Python 3.12, SQLAlchemy 2, Alembic, FastAPI, pytest (SQLite + PostgreSQL scratch).

**Spec:** `docs/superpowers/specs/2026-09-22-ltt-workflow-platform-v2-design.md` (§3, §4.1–4.3, §5.1–5.8, §5.11, §7, §8.1, §16, §18). Ledger: `docs/superpowers/plans/2026-09-22-wf-v2-LEDGER.md`.

"""Nhật ký cấp run — `workflow_event`, CHỈ THÊM (spec §5.11).

Cấp node KHÔNG ghi ở đây: `node_run_snapshot` đã là nhật ký của node.

⚠️ `seq` = max + 1. Bên gọi phải đang giữ khoá hàng `workflow_run` (động cơ)
   hoặc là bên vừa tạo run (`tao_run`); `UNIQUE(run_id, seq)` là chốt cuối.
"""
from __future__ import annotations

from typing import Optional

from sqlalchemy import func, select

from platform_core import job_events
from platform_core.models import WorkflowEvent

LOAI = frozenset({
    "WorkflowStarted", "TriggerReceived", "NodeResumed", "WaitpointOpened",
    "WaitpointSignalled", "BudgetChecked", "SupervisorDecision",
    "WorkflowCompleted", "WorkflowFailed", "WorkflowCancelled",
})


def ghi(db, run_id: str, loai: str, *, node_id: str = "",
        du_lieu: Optional[dict] = None) -> WorkflowEvent:
    if loai not in LOAI:
        raise ValueError(f"loai su kien khong hop le: {loai!r}")
    cuoi = db.execute(select(func.max(WorkflowEvent.seq))
                      .where(WorkflowEvent.run_id == run_id)).scalar()
    ev = WorkflowEvent(run_id=run_id, seq=int(cuoi or 0) + 1, loai=loai,
                       node_id=node_id or "",
                       du_lieu=job_events.redact_details(dict(du_lieu or {})))
    db.add(ev)
    db.flush()
    return ev


def doc(db, run_id: str, *, sau: int = 0) -> list[WorkflowEvent]:
    return list(db.execute(
        select(WorkflowEvent)
        .where(WorkflowEvent.run_id == run_id, WorkflowEvent.seq > int(sau))
        .order_by(WorkflowEvent.seq)).scalars())


__all__ = ["LOAI", "ghi", "doc"]
```

"""Một chỗ tra ToolSpec cho workflow: tên → (app, job_type, version, giá).

Dùng lại `platform_core.agent.tools.discover_tools` — không quét manifest lần
hai. Plan 02 dựng `NodeDefinition` (spec §4.4) TRÊN mô-đun này.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable

from platform_core import registry


@dataclass(frozen=True)
class CongCu:
    ten: str
    app_slug: str
    job_type: str
    type_version: int
    params_schema: dict
    side_effect: str
    idempotency_fields: tuple[str, ...]
    tinh_gia: Callable[[dict], int]


def tat_ca() -> dict[str, CongCu]:
    ra: dict[str, CongCu] = {}
    for item in registry.discover():
        if item.error or not item.manifest.enabled:
            continue
        for t in item.manifest.tools:
            jt = t.job_type
            ra[t.name] = CongCu(
                ten=t.name, app_slug=item.slug, job_type=jt,
                type_version=int(getattr(t, "type_version", 1) or 1),
                params_schema=dict(t.params_schema or {}),
                side_effect=str(getattr(t, "side_effect", "external")
                                or "external"),
                idempotency_fields=tuple(
                    (getattr(t, "idempotency", None) or {}).get("key_fields", ())),
                tinh_gia=lambda p, _jt=jt: int(registry.credit_for(_jt, p)))
    return ra


__all__ = ["CongCu", "tat_ca"]

_ID = re.compile(r"^[A-Za-z0-9_\-]{1,64}$")
_CU = re.compile(r"^\$([A-Za-z0-9_\-]{1,64})\.([A-Za-z0-9_\-]{1,64})$")
_THAM_CHIEU = re.compile(
    r"^\{\{\s*nodes\.([A-Za-z0-9_\-]{1,64})\.output\.([A-Za-z0-9_\-]{1,64})\s*\}\}$")
_KIND = frozenset({"DATA", "CONTROL", "ERROR"})

class LoiBienDich(ValueError):
    def __init__(self, ma: str, thong_diep: str, node_id: str = ""):
        super().__init__(f"{ma}: {thong_diep}")
        self.ma, self.node_id = ma, node_id


def la_tham_chieu(gia_tri: Any) -> Optional[tuple[str, str]]:
    if isinstance(gia_tri, str):
        m = _THAM_CHIEU.match(gia_tri.strip())
        if m:
            return m.group(1), m.group(2)
    return None


def _doi(gia_tri: Any, node_id: str) -> Any:
    """`$a.b` (Canvas cũ) → `{{ nodes.a.output.b }}`; chặn `{{ }}` khác."""
    if isinstance(gia_tri, dict):
        return {k: _doi(v, node_id) for k, v in gia_tri.items()}
    if isinstance(gia_tri, list):
        return [_doi(v, node_id) for v in gia_tri]
    if isinstance(gia_tri, str):
        m = _CU.match(gia_tri.strip())
        if m:
            return "{{ nodes.%s.output.%s }}" % (m.group(1), m.group(2))
        if "{{" in gia_tri and la_tham_chieu(gia_tri) is None:
            raise LoiBienDich("bieu_thuc_chua_ho_tro",
                              "chi ho tro {{ nodes.<id>.output.<cong> }}",
                              node_id)
    return gia_tri


def _tham_chieu_trong(gia_tri: Any) -> set[str]:
    if isinstance(gia_tri, dict):
        return set().union(*(_tham_chieu_trong(v) for v in gia_tri.values())) \
            if gia_tri else set()
    if isinstance(gia_tri, list):
        return set().union(*(_tham_chieu_trong(v) for v in gia_tri)) \
            if gia_tri else set()
    tc = la_tham_chieu(gia_tri)
    return {tc[0]} if tc else set()

    nid = str(n.get("id", ""))
    if not _ID.match(nid):
        raise LoiBienDich("id_khong_hop_le", f"id node {nid[:70]!r}", nid[:64])
    loai = str(n.get("type", ""))
    if loai == "tool":
        data = n.get("data") or {}
        ten, params = str(data.get("tool", "")), data.get("params") or {}
        settings = data.get("settings") or {}
    elif loai.startswith("app."):
        ten, params = loai[4:], n.get("params") or {}
        settings = n.get("settings") or {}
    else:
        raise LoiBienDich("loai_node_chua_ho_tro", loai, nid)
    tool = cc.get(ten)
    if tool is None:
        raise LoiBienDich("node_khong_ton_tai", ten, nid)

            "params": _doi(dict(params), nid),
            "credential_ref": str(n.get("credential_ref") or ""),
            "settings": dict(settings), "thi_hanh": "job"}


def bien_dich(document: dict, *,
              cong_cu: Optional[dict[str, CongCu]] = None) -> dict:
    # Gọi qua MÔ-ĐUN (không `from … import tat_ca`) để monkeypatch trong test
    # thay được nguồn — bẫy đã thấy khi viết plan.
    cc = _nguon.tat_ca() if cong_cu is None else cong_cu
    nodes: list[dict] = []
    edges: list[dict] = []
    trang_cua: dict[str, str] = {}
    for trang in document.get("pages") or []:
        pid = str(trang.get("id", ""))
        for n in trang.get("nodes") or []:
            nd = _node(n, cc)
            if nd["id"] in trang_cua:
                raise LoiBienDich("id_trung", nd["id"], nd["id"])
            trang_cua[nd["id"]] = pid
            nodes.append(nd)
        for c in trang.get("edges") or []:
            e = _canh_v3(c)
            e["kind"] = str(c.get("kind") or "DATA")
            e["condition"] = str(c.get("condition") or "")
            edges.append((pid, e))  # type: ignore[arg-type]
    if not nodes:
        raise LoiBienDich("do_thi_rong", "khong co node nao")

    ra_edges: list[dict] = []
    for pid, e in edges:  # type: ignore[misc]
        a, b = e["source_node"], e["target_node"]
        if a not in trang_cua or b not in trang_cua:
            raise LoiBienDich("canh_khong_hop_le", f"{a}->{b}", b)
        if e["kind"] not in _KIND:
            raise LoiBienDich("canh_khong_hop_le", e["kind"], b)
        if trang_cua[a] != pid or trang_cua[b] != pid:
            raise LoiBienDich("canh_khac_trang", f"{a}->{b}", b)
        ra_edges.append(e)

    # Kahn: thứ tự tô-pô + phát hiện chu trình.
    vao = {n["id"]: 0 for n in nodes}
    ke: dict[str, list[str]] = {n["id"]: [] for n in nodes}
    for e in ra_edges:
        ke[e["source_node"]].append(e["target_node"])
        vao[e["target_node"]] += 1
    q = deque(sorted(k for k, v in vao.items() if v == 0))
    thu_tu: list[str] = []
    while q:
        k = q.popleft()
        thu_tu.append(k)
        for t in sorted(ke[k]):
            vao[t] -= 1
            if vao[t] == 0:
                q.append(t)
    if len(thu_tu) != len(nodes):
        raise LoiBienDich("chu_trinh", "do thi co chu trinh")

    to_tien: dict[str, set[str]] = {k: set() for k in thu_tu}
    for k in thu_tu:
        for t in ke[k]:
            to_tien[t] |= to_tien[k] | {k}
    for n in nodes:
        for dich in _tham_chieu_trong(n["params"]):
            if dich not in to_tien[n["id"]]:
                raise LoiBienDich("tham_chieu_khong_phai_to_tien", dich, n["id"])

    cai_dat = document.get("settings") or {}
    return {
        "schema": SCHEMA, "inputs": [], "outputs": [],
        "settings": {
            "max_concurrency": max(1, min(_MAX_CONCURRENCY,
                                          int(cai_dat.get("max_concurrency", 4)))),
            "timeout_s": int(cai_dat.get("timeout_s", _TIMEOUT_MAC_DINH)),
        },
        "nodes": sorted(nodes, key=lambda n: n["id"]),
        "edges": sorted(ra_edges, key=lambda e: e["id"]),
        "thu_tu": thu_tu,
    }


__all__ = ["SCHEMA", "LoiBienDich", "bien_dich", "la_tham_chieu"]

import itertools
import math

from platform_core.workflow import cong_cu as _nguon
from platform_core.workflow.bien_dich import la_tham_chieu

def _ung_vien(khoa: str, schema: dict) -> list:
    muc = schema.get(khoa) or {}
    kieu = str(muc.get("type", ""))
    if kieu == "enum" and isinstance(muc.get("values"), list) and muc["values"]:
        return list(muc["values"])
    if kieu in ("int", "number") and muc.get("max") is not None:
        return [muc["max"]]
    return []            # không biên ⇒ tính theo mặc định (bỏ khoá)


def _gia_node(node: dict, cc: dict[str, CongCu]) -> int:
    if node.get("thi_hanh") != "job":
        return 0
    tool = cc[str(node["type"])[4:]]
    params = dict(node.get("params") or {})
    dong = [k for k, v in params.items()
            if la_tham_chieu(v) is not None
            or (isinstance(v, str) and "{{" in v)]
    co_dinh = {k: v for k, v in params.items() if k not in dong}
    truc = [(k, _ung_vien(k, tool.params_schema)) for k in dong]
    truc = [(k, vs) for k, vs in truc if vs]
    to_hop = itertools.product(*(vs for _k, vs in truc)) if truc else [()]
    cao = 0
    for bo in to_hop:
        thu = dict(co_dinh)
        thu.update({k: v for (k, _vs), v in zip(truc, bo)})
        try:
            cao = max(cao, int(tool.tinh_gia(thu)))
        except Exception as loi:                      # noqa: BLE001
            raise KhongBaoGiaDuoc(f"{node['id']}: {loi}") from loi
    return cao


def bao_gia_do_thi(graph: dict, *,
                   cong_cu: dict[str, CongCu] | None = None) -> BaoGia:
    """Trần tất định của cả đồ thị. GHI `gia_tran` từng node + `bang_gia`."""
    cc = _nguon.tat_ca() if cong_cu is None else cong_cu
    bang: dict[str, int] = {}
    for node in graph.get("nodes") or []:
        gia = _gia_node(node, cc)
        node["gia_tran"] = gia
        bang[str(node["id"])] = gia
    graph["bang_gia"] = bang
    toi_da = sum(bang.values())
    return BaoGia(toi_da=toi_da,
                  uoc_tinh=int(math.ceil(toi_da * _TY_LE_UOC_TINH)))

"""Cửa tạo run DUY NHẤT — API, Canvas, trigger đều đi qua đây (spec §8.1).

Một transaction: ghim version → hold một lần → CREATED→QUEUED → wakeup `start`
→ sự kiện `WorkflowStarted` → đánh thức động cơ. Không commit.
"""
from __future__ import annotations

from datetime import datetime, timezone

from sqlalchemy import func, select
from sqlalchemy.exc import IntegrityError

from platform_core import jobs
from platform_core.job_scheduler import ORIGIN_WORKFLOW
from platform_core.models import Workflow, WorkflowRun, WorkflowVersion
from platform_core.workflow import (danh_thuc, may_trang_thai, su_kien,
                                    thi_hanh, tien)

DONG_CO_JOB = "workflow.dong_co"
DONG_CO_USER = "__workflow_dong_co__"
#: Hạn mức (spec §7.3). Đọc từ settings ở plan 06; hằng ở đây là mặc định spec.
MAX_RUN_DONG_THOI = 5
_DANG_SONG = ("CREATED", "QUEUED", "RUNNING", "WAITING")


class KhongCoBanPublish(ValueError):
    pass


class QuaHanMuc(ValueError):
    pass


class TrungKhoaKhacWorkflow(ValueError):
    pass


def _bay_gio() -> datetime:
    return datetime.now(timezone.utc).replace(tzinfo=None)


def danh_thuc_dong_co(db) -> None:
    """Nộp job động cơ. `jobs.submit` khử trùng job CHƯA kết thúc cùng khoá ⇒
    tối đa một job động cơ chờ trên toàn hệ."""
    jobs.submit(db, user_id=DONG_CO_USER, app_slug="agent",
                job_type=DONG_CO_JOB, params={}, dedupe_key="workflow:dong_co",
                max_attempts=3, origin=ORIGIN_WORKFLOW,
                priority_class="internal")


def _tim(db, owner_id: str, khoa: str):
    return db.execute(select(WorkflowRun).where(
        WorkflowRun.owner_id == owner_id,
        WorkflowRun.idempotency_key == khoa)).scalars().first()


def tao_run(db, *, workflow: Workflow, owner_id: str, trigger_kind: str,
            idempotency_key: str, inputs: dict | None = None
            ) -> tuple[WorkflowRun, bool]:
    if workflow.owner_id != owner_id:
        raise KhongCoBanPublish("workflow khong thuoc nguoi goi")
    if not workflow.published_version_id or workflow.archived_at is not None:
        raise KhongCoBanPublish("workflow chua publish hoac da luu tru")
    ver = db.get(WorkflowVersion, workflow.published_version_id)
    if ver is None:
        raise KhongCoBanPublish("version publish khong ton tai")

    if idempotency_key:
        cu = _tim(db, owner_id, idempotency_key)
        if cu is not None:
            if cu.workflow_id != workflow.id:
                raise TrungKhoaKhacWorkflow(idempotency_key)
            return cu, False

    dang = db.execute(select(func.count()).select_from(WorkflowRun).where(
        WorkflowRun.owner_id == owner_id,
        WorkflowRun.status.in_(_DANG_SONG))).scalar() or 0
    if dang >= MAX_RUN_DONG_THOI:
        raise QuaHanMuc(f"toi da {MAX_RUN_DONG_THOI} run dong thoi")

    try:
        with db.begin_nested():
            run = thi_hanh.tao_run(
                db, workflow_id=workflow.id, version_id=ver.id,
                owner_id=owner_id, trigger_kind=trigger_kind,
                idempotency_key=idempotency_key,
                estimated_max=int(ver.estimated_max or 0))
    except IntegrityError:
        cu = _tim(db, owner_id, idempotency_key)
        if cu is None:
            raise
        if cu.workflow_id != workflow.id:
            raise TrungKhoaKhacWorkflow(idempotency_key) from None
        return cu, False

    tien.giu_cho_run(db, run)                     # ném ThieuCreditError
    may_trang_thai.chuyen_run(db, run.id, sang="QUEUED")
    danh_thuc.hen(db, run_id=run.id, kind="start", due_at=_bay_gio())
    su_kien.ghi(db, run.id, "WorkflowStarted",
                du_lieu={"trigger_kind": trigger_kind,
                         "version_id": ver.id,
                         "inputs": dict(inputs or {})})
    danh_thuc_dong_co(db)
    return run, True


__all__ = ["DONG_CO_JOB", "DONG_CO_USER", "MAX_RUN_DONG_THOI",
           "KhongCoBanPublish", "QuaHanMuc", "TrungKhoaKhacWorkflow",
           "danh_thuc_dong_co", "tao_run"]
```

class LoiThamSo(ValueError):
    def __init__(self, ma: str, thong_diep: str):
        super().__init__(f"{ma}: {thong_diep}")
        self.ma = ma


def _bay_gio():
    from datetime import datetime, timezone
    return datetime.now(timezone.utc).replace(tzinfo=None)


def giai_tham_so(db, run_id: str, params: dict) -> dict:
    def _giai(v: Any) -> Any:
        if isinstance(v, dict):
            return {k: _giai(x) for k, x in v.items()}
        if isinstance(v, list):
            return [_giai(x) for x in v]
        tc = la_tham_chieu(v)
        if tc is None:
            return v
        nguon, cong = tc
        anh = thi_hanh.anh_chup_hien_tai(db, run_id, nguon)
        if anh is None or anh.execution_status != "SUCCEEDED":
            raise LoiThamSo("dau_vao_sai", f"{nguon} chua SUCCEEDED")
        if cong not in (anh.output_ref or {}):
            raise LoiThamSo("dau_vao_sai", f"{nguon} khong co cong {cong!r}")
        return anh.output_ref[cong]
    return _giai(dict(params))


def _khoa_idem(run: WorkflowRun, node: dict, lan: int, cc: CongCu,
               params: dict) -> str:
    if cc.side_effect == "none" or not cc.idempotency_fields:
        goc = [run.id, node["id"], lan]
    else:
        goc = [run.id, node["id"],
               {k: params.get(k) for k in cc.idempotency_fields}]
    return hashlib.sha256(json.dumps(goc, sort_keys=True, default=str)
                          .encode()).hexdigest()[:64]


def nop_job(db, run: WorkflowRun, node: dict, *,
            cong_cu: Optional[dict[str, CongCu]] = None) -> Optional[str]:
    cc_all = _nguon.tat_ca() if cong_cu is None else cong_cu
    nid = node["id"]
    cc = cc_all[str(node["type"])[4:]]
    anh = thi_hanh.anh_chup_hien_tai(db, run.id, nid)
    lan = int(anh.attempt_number if anh else 1)
    try:
        params = giai_tham_so(db, run.id, node.get("params") or {})
    except LoiThamSo as loi:
        may_trang_thai.chuyen_node(db, run.id, nid, sang="FAILED",
                                   mo_ta="giai tham so", error=f"{loi.ma}: {loi}",
                                   attempt_number=lan)
        return None
    gia_that = int(cc.tinh_gia(params))
    tran = int(node.get("gia_tran", 0))
    if gia_that > tran:
        may_trang_thai.chuyen_node(
            db, run.id, nid, sang="FAILED", mo_ta="vuot tran bao gia",
            error=f"vuot_tran_bao_gia: {gia_that} > {tran}", attempt_number=lan)
        return None
    wp = cho_doi.tao_waitpoint(db, run_id=run.id, owner_id=run.owner_id,
                               kind="JOB", required_successes=1)
    cho_doi.chan_node(db, run_id=run.id, node_id=nid, waitpoint_id=wp.id)
    tham_so = dict(params)
    tham_so["owner_id"] = run.owner_id           # như submit_child (v1)
    khoa = _khoa_idem(run, node, lan, cc, params)
    tham_so["_idem"] = khoa
    job = jobs.submit(db, user_id=run.owner_id, app_slug=cc.app_slug,
                      job_type=cc.job_type, params=tham_so, dedupe_key=khoa,
                      max_attempts=3, parent_id=wp.id, plan_id=run.id,
                      origin=ORIGIN_WORKFLOW, priority_class="internal")
    may_trang_thai.chuyen_node(
        db, run.id, nid, sang="QUEUED", mo_ta="da nop job", job_id=job.id,
        attempt_number=lan, output_ref={"_gia_that": gia_that})
    return job.id


def bao_ket_thuc(db, job: Job) -> None:

def ap_ket_qua(db, run: WorkflowRun, node_id: str, *,
               cong_cu: Optional[dict[str, CongCu]] = None) -> Optional[str]:
    hien = may_trang_thai.trang_thai_node(db, run.id, node_id)
    if hien not in ("QUEUED", "RUNNING"):
        return None
    for wp_id in cho_doi.waitpoint_cua_node(db, run.id, node_id):
        wp = db.get(Waitpoint, wp_id)
        if wp is None or wp.kind != "JOB" or wp.status != "PENDING":
            continue
        sig = db.execute(select(WaitpointSignal).where(
            WaitpointSignal.waitpoint_id == wp_id)).scalars().first()
        if sig is None:
            continue
        wp.status = "COMPLETED"
        anh = thi_hanh.anh_chup_hien_tai(db, run.id, node_id)
        lan = int(anh.attempt_number if anh else 1)
        gia_that = int((anh.output_ref or {}).get("_gia_that", 0)) if anh else 0
        if sig.status == "SUCCEEDED":
            du_lieu = dict(sig.result_ref or {})
            if len(json.dumps(du_lieu, default=str)) > GIOI_HAN_DAU_RA:
                may_trang_thai.chuyen_node(
                    db, run.id, node_id, sang="FAILED", mo_ta="dau ra",
                    error="dau_ra_qua_lon", attempt_number=lan, job_id=sig.ref_id)
                return "FAILED"
            bao = du_lieu.get("credit_cost")      # handler báo giá thật (vd theo token)
            chi_phi = int(bao) if bao is not None else gia_that
            node = _node_cua(db, run, node_id)
            chi_phi = min(chi_phi, int(node.get("gia_tran", chi_phi)))  # không thu quá trần
            may_trang_thai.chuyen_node(
                db, run.id, node_id, sang="SUCCEEDED", mo_ta="job xong",
                output_ref=du_lieu, cost_credit=chi_phi, attempt_number=lan,
                job_id=sig.ref_id)
            run.actual_cost = int(run.actual_cost or 0) + chi_phi
            db.flush()
            return "SUCCEEDED"
        mo_ho = (sig.result_ref or {}).get("provider_state") == _MO_HO
        cc_all = _nguon.tat_ca() if cong_cu is None else cong_cu
        se = cc_all[str(_node_cua(db, run, node_id)["type"])[4:]].side_effect
        sang = "ABANDONED" if (mo_ho and se == "irreversible") else (
            "CANCELLED" if sig.status == "CANCELLED" else "FAILED")
        ma = "ket_qua_mo_ho" if sang == "ABANDONED" else "provider_loi"
        may_trang_thai.chuyen_node(
            db, run.id, node_id, sang=sang, mo_ta="job ket thuc",
            error=f"{ma}: {(sig.result_ref or {}).get('error', '')}"[:2000],
            attempt_number=lan, job_id=sig.ref_id)
        return sang
    return None

"""Động cơ Workflow V2 — MỘT runtime (spec §5.1).

Không I/O: chỉ đọc/ghi CSDL. Việc có mạng/GPU/tiền đi thành job (`cau_noi`).
`mot_vong` mở MỖI RUN MỘT TRANSACTION, khoá hàng run `SKIP LOCKED` — nhiều
worker chạy job động cơ cùng lúc vẫn an toàn.
"""
from __future__ import annotations

from datetime import datetime, timedelta, timezone
from typing import Callable, Optional

from sqlalchemy import select

from platform_core import jobs
from platform_core.models import Job, Wakeup, WorkflowRun, WorkflowVersion
from platform_core.workflow import cau_noi, may_trang_thai as mtt, su_kien
from platform_core.workflow.cong_cu import CongCu

#: Trần cứng — PHẢI ngắn hơn `credit.TU_HOAN_SAU` (lập luận ở `dieu_phoi` cũ,
#: `test_965` ghim quan hệ này).
TRAN_RUN = timedelta(hours=20)
_KICH = {"DATA": frozenset({"SUCCEEDED"}),
         "ERROR": frozenset({"FAILED", "ABANDONED"})}
_CAI_CACHE: dict[str, dict] = {}


def _bay_gio() -> datetime:
    return datetime.now(timezone.utc).replace(tzinfo=None)


def _do_thi(db, version_id: str) -> dict:
    g = _CAI_CACHE.get(version_id)
    if g is None:
        g = db.get(WorkflowVersion, version_id).graph   # bất biến ⇒ cache an toàn
        if len(_CAI_CACHE) > 512:
            _CAI_CACHE.clear()
        _CAI_CACHE[version_id] = g
    return g


def _lan_truyen(db, run: WorkflowRun, g: dict) -> dict[str, str]:
    tt = {n["id"]: mtt.trang_thai_node(db, run.id, n["id"]) for n in g["nodes"]}
    vao: dict[str, list[dict]] = {n["id"]: [] for n in g["nodes"]}
    for e in g["edges"]:
        vao[e["target_node"]].append(e)
    doi = True
    while doi:
        doi = False
        for nid in g["thu_tu"]:
            if tt[nid] != "PENDING":
                continue
            canh = vao[nid]
            kich = [tt[e["source_node"]] in _KICH.get(e["kind"], frozenset())
                    for e in canh]
            chet = [tt[e["source_node"]] in mtt.NODE_KET_THUC and not k
                    for e, k in zip(canh, kich)]
            if any(chet):
                mtt.chuyen_node(db, run.id, nid, sang="SKIPPED",
                                mo_ta="nguon khong kich hoat")
                tt[nid] = "SKIPPED"
                doi = True
            elif all(kich):
                mtt.chuyen_node(db, run.id, nid, sang="READY", mo_ta="du dau vao")
                tt[nid] = "READY"
                doi = True
    return tt


def _ket_run(db, run: WorkflowRun, g: dict, tt: dict[str, str]) -> Optional[str]:
    if not all(s in mtt.NODE_KET_THUC for s in tt.values()):
        return None
    co_nhanh_loi = {e["source_node"] for e in g["edges"] if e["kind"] == "ERROR"}
    hong = [n for n, s in tt.items()
            if s in ("FAILED", "ABANDONED") and n not in co_nhanh_loi]
    sang = "FAILED" if hong else "SUCCEEDED"
    if run.status == "QUEUED":
        mtt.chuyen_run(db, run.id, sang="RUNNING")
    su_kien.ghi(db, run.id, "WorkflowFailed" if hong else "WorkflowCompleted",
                du_lieu={"status": sang, "node_hong": hong})
    mtt.chuyen_run(db, run.id, sang=sang, ly_do="moi node ket thuc")
    return sang


def xu_ly_run(db, run_id: str, *,
              cong_cu: Optional[dict[str, CongCu]] = None) -> str:
    run = db.get(WorkflowRun, run_id)
    if run is None or run.status in mtt.RUN_KET_THUC:
        return run.status if run else "MISSING"
    g = _do_thi(db, run.version_id)
    if run.status == "QUEUED":
        mtt.chuyen_run(db, run.id, sang="RUNNING")
    for n in g["nodes"]:
        cau_noi.ap_ket_qua(db, run, n["id"], cong_cu=cong_cu)
    tt = _lan_truyen(db, run, g)
    ket = _ket_run(db, run, g, tt)
    if ket:
        return ket
    dang = sum(1 for s in tt.values() if s in mtt.NODE_DANG_CHAY)
    tran = int(g["settings"].get("max_concurrency", 4))
    for nid in g["thu_tu"]:
        if dang >= tran:
            break
        if tt[nid] != "READY":
            continue
        node = next(n for n in g["nodes"] if n["id"] == nid)
        if cau_noi.nop_job(db, run, node, cong_cu=cong_cu):
            dang += 1
    tt = _lan_truyen(db, run, g)            # node FAILED lúc nộp ⇒ lan tiếp
    return _ket_run(db, run, g, tt) or run.status


def huy_run(db, run_id: str, ly_do: str) -> bool:
    run = db.get(WorkflowRun, run_id)
    if run is None or run.status in mtt.RUN_KET_THUC:
        return False
    g = _do_thi(db, run.version_id)
    for n in g["nodes"]:
        if mtt.trang_thai_node(db, run.id, n["id"]) not in mtt.NODE_KET_THUC:
            mtt.chuyen_node(db, run.id, n["id"], sang="CANCELLED", mo_ta=ly_do)
    for job in db.execute(select(Job).where(Job.plan_id == run.id)).scalars():
        jobs.cancel(db, job)                 # chỉ đổi job queued/running
    su_kien.ghi(db, run.id, "WorkflowCancelled", du_lieu={"ly_do": ly_do})
    return mtt.chuyen_run(db, run.id, sang="CANCELLED", ly_do=ly_do)


def ttl(db) -> int:
    dem = 0
    luc = _bay_gio()
    for run in db.execute(select(WorkflowRun).where(
            WorkflowRun.status.in_(("QUEUED", "RUNNING", "WAITING")),
            WorkflowRun.started_at.is_not(None))).scalars():
        g = _do_thi(db, run.version_id)
        han = min(timedelta(seconds=int(g["settings"].get("timeout_s", 21600))),
                  TRAN_RUN)
        if run.started_at <= luc - han:
            for job in db.execute(select(Job).where(Job.plan_id == run.id)).scalars():
                jobs.cancel(db, job)
            su_kien.ghi(db, run.id, "WorkflowFailed",
                        du_lieu={"status": "TIMED_OUT"})
            if mtt.chuyen_run(db, run.id, sang="TIMED_OUT", ly_do="qua han"):
                dem += 1
    return dem


def mot_vong(db_factory: Callable, *, gioi_han: int = 50) -> dict:
    db = db_factory()
    try:
        run_ids = list(dict.fromkeys(db.execute(
            select(Wakeup.run_id).where(Wakeup.claimed_at.is_(None),
                                        Wakeup.due_at <= _bay_gio())
            .order_by(Wakeup.due_at).limit(gioi_han * 4)).scalars()))[:gioi_han]
        so = ttl(db)
        db.commit()
    finally:
        db.close()
    da = 0
    for rid in run_ids:
        db = db_factory()
        try:
            run = db.execute(select(WorkflowRun).where(WorkflowRun.id == rid)
                             .with_for_update(skip_locked=True)).scalars().first()
            if run is None:
                db.rollback()
                continue
            ws = list(db.execute(select(Wakeup).where(
                Wakeup.run_id == rid, Wakeup.claimed_at.is_(None),
                Wakeup.due_at <= _bay_gio()).with_for_update(skip_locked=True))
                .scalars())
            xu_ly_run(db, rid)
            for w in ws:
                w.claimed_at = _bay_gio()
            db.commit()
            da += 1
        except Exception:
            db.rollback()
            raise
        finally:
            db.close()
    db = db_factory()
    try:
        ke = db.execute(select(Wakeup.due_at).where(Wakeup.claimed_at.is_(None))
                        .order_by(Wakeup.due_at).limit(1)).scalar()
        con = db.execute(select(WorkflowRun.id).where(WorkflowRun.status.in_(
            ("QUEUED", "RUNNING", "WAITING"))).limit(1)).scalar() is not None
    finally:
        db.close()
    return {"run_da_xu_ly": da, "timed_out": so,
            "con_viec": bool(ke is not None or con), "han_ke_tiep": ke}


__all__ = ["TRAN_RUN", "mot_vong", "xu_ly_run", "huy_run", "ttl"]
