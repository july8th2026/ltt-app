```python
"""`ToolSpec.params_schema` (từ vựng LTT) → tập con JSON Schema 2020-12 (§4.6).

🔴 **Hai từ vựng, và chỉ tệp này biết cả hai.** `params_schema` của LTT dùng

   `he_kieu.xuat_luat_noi_day()` sinh ra để tránh.

from __future__ import annotations

from typing import Any

from platform_core.workflow import he_kieu


class LoiLuocDo(ValueError):
    """Từ vựng thiếu một dòng. ⛔ Không bao giờ bỏ qua im lặng — một tham số
    biến mất khỏi form trông y hệt một tham số không tồn tại."""


#: Kiểu tham số LTT → kiểu **cổng** của `he_kieu`.
KIEU_CONG: dict[str, str] = {
    "string": "text", "text": "text", "enum": "text",
    "int": "integer", "number": "number", "bool": "boolean",
    "model_ref": "model_ref",
    "image": "image", "image[]": "image[]",
    "video": "video", "video[]": "video[]",
    "audio": "audio",
    "json": "json", "json[]": "json[]",
}

#: Kiểu tham số LTT → khung JSON Schema. `enum` xử lý riêng (kiểu suy từ giá trị).
_KHUNG: dict[str, dict] = {
    "string": {"type": "string"},
    "text": {"type": "string", "x-ltt-control": "textarea"},
    "int": {"type": "integer"},
    "number": {"type": "number"},
    "bool": {"type": "boolean"},
    "model_ref": {"type": "string", "x-ltt-kieu": "model_ref",
                  "x-ltt-control": "dropdown-dong"},
    "image": {"type": "string", "x-ltt-kieu": "image"},
    "image[]": {"type": "array", "items": {"type": "string"},
                "x-ltt-kieu": "image[]"},
    "video": {"type": "string", "x-ltt-kieu": "video"},
    "video[]": {"type": "array", "items": {"type": "string"},

    "audio": {"type": "string", "x-ltt-kieu": "audio"},
    "json": {"type": "object", "x-ltt-kieu": "json"},
    "json[]": {"type": "array", "items": {"type": "object"},
               "x-ltt-kieu": "json[]"},
}


def _la_may_tu_dien(rule: dict) -> bool:
    """Nền tảng tự điền (ffprobe qua `measured_from`) ⇒ không phải ô nhập, và
    KHÔNG phải cổng: không ai nối được vào một giá trị server tự đo."""
    return bool(rule.get("measured_from"))


def mot_tham_so(ten: str, rule: dict) -> dict:
    kieu = str(rule.get("type", ""))
    if kieu == "enum":
        gia_tri = list(rule.get("values") or [])
        # Kiểu lấy từ CHÍNH giá trị: `target_height` là enum số.
        json_kieu = ("integer" if gia_tri and all(
            isinstance(v, int) and not isinstance(v, bool) for v in gia_tri)
            else "number" if gia_tri and all(
                isinstance(v, (int, float)) and not isinstance(v, bool)
                for v in gia_tri)
            else "string")
        ra: dict[str, Any] = {"type": json_kieu, "enum": gia_tri}
    else:
        khung = _KHUNG.get(kieu)
        if khung is None:
            raise LoiLuocDo(
                f"kieu tham so chua co tu vung: {kieu!r} (tham so {ten!r}); "
                f"them mot dong vao luoc_do._KHUNG thay vi doan")
        ra = {k: (dict(v) if isinstance(v, dict) else v)
              for k, v in khung.items()}

    if "min" in rule:
        ra["minimum"] = rule["min"]
    if "max" in rule:
        ra["maximum"] = rule["max"]
    if "max_length" in rule:
        ra["maxLength"] = int(rule["max_length"])
    if "default" in rule:
        ra["default"] = rule["default"]
    if rule.get("description"):
        ra["description"] = str(rule["description"])
    # `values`/`options` đã được `agent.tools.resolve_params_schema` giải sẵn —
    # `model_ref`, `source` VÀ `ToolSpec.tuy_chon_dong` (enum theo NGƯỜI DÙNG)
    # đều ghi vào `values`. Mang sang JSON Schema, không giải lại.
    # ⛔ `"values" in rule`, KHÔNG `rule.get("values")`: danh sách rỗng nghĩa là
    #    "bạn chưa có lựa chọn nào", khác hẳn "gõ gì cũng được". Bỏ `enum` khi
    #    rỗng là trả lại đúng ô chữ tự do mà GENERIC_GAP D10 sinh ra để đóng.
    if kieu != "enum" and "values" in rule:
        ra["enum"] = list(rule["values"] or [])
    if rule.get("options"):
        ra["x-ltt-nhan"] = {str(o["value"]): str(o.get("label", o["value"]))
                            for o in rule["options"] if isinstance(o, dict)}
    if rule.get("depends_on"):
        # `depends_on` = tham số phụ thuộc MỘT THAM SỐ KHÁC của cùng node (vd
        # `taovideo`: `resolution` hợp lệ tuỳ `model`), giải qua
        # `providers.model_capabilities`. ⛔ KHÔNG phải `ToolSpec.tuy_chon_dong`
        # — cơ chế đó đi qua nhánh `values` ở trên và ra thẳng `enum`.
        # ⛔ `"nguon"` là MARKER gửi client, không phải tên trường nội bộ.
        ra["x-ltt-dong"] = {"nguon": "app_property",
                            "prop": str(rule["depends_on"])}
    if _la_may_tu_dien(rule):
        ra["readOnly"] = True
        ra["x-ltt-control"] = "may_tu_dien"
    return ra


def sang_json_schema(params_schema: dict) -> dict:
    thuoc_tinh: dict[str, dict] = {}
    bat_buoc: list[str] = []
    for ten, rule in (params_schema or {}).items():
        rule = rule if isinstance(rule, dict) else {}
        thuoc_tinh[ten] = mot_tham_so(ten, rule)
        if rule.get("required") and not _la_may_tu_dien(rule):
            bat_buoc.append(ten)
    return {"type": "object", "properties": thuoc_tinh,
            "required": bat_buoc, "additionalProperties": False}


def _cong(kieu: str, *, bat_buoc: bool, nhan: str) -> he_kieu.Cong:
    if not he_kieu.hop_le(kieu):
        raise LoiLuocDo(f"kieu cong khong hop le: {kieu!r} (cong {nhan!r})")
    return he_kieu.Cong(type=kieu, required=bat_buoc, multiple=False, nhan=nhan)


def cong_vao(params_schema: dict, cong_khai: dict) -> tuple:
    ra: list[tuple[str, he_kieu.Cong]] = []
    for ten, rule in (params_schema or {}).items():
        rule = rule if isinstance(rule, dict) else {}
        if _la_may_tu_dien(rule):
            continue
        kieu = str((cong_khai or {}).get(ten) or
                   KIEU_CONG.get(str(rule.get("type", "")), ""))
        if not kieu:
            raise LoiLuocDo(
                f"kieu tham so chua co tu vung: {rule.get('type')!r} "
                f"(tham so {ten!r})")
        ra.append((ten, _cong(kieu, bat_buoc=bool(rule.get("required")),
                              nhan=ten)))
    return tuple(ra)
```

```python
@dataclass(frozen=True)
class Gia:
    """`mien_phi` | `co_dinh(n)` | `theo_registry(job_type)` (spec §7.1)."""

    kieu: str
    so: int = 0
    job_type: str = ""

    @staticmethod
    def mien_phi() -> "Gia":
        return Gia("mien_phi")

    @staticmethod
    def co_dinh(n: int) -> "Gia":
        return Gia("co_dinh", so=int(n))

    @staticmethod
    def theo_registry(job_type: str) -> "Gia":
        return Gia("theo_registry", job_type=str(job_type))

    def sang_json(self) -> dict:
        return {"kieu": self.kieu, "so": self.so, "job_type": self.job_type}

class NodeDefinition:
    type: str
    type_version: int
    kind: str
    category: str
    title: str
    description: str
    icon: str
    cong_vao: tuple
    cong_ra: tuple
    tham_so: dict
    thi_hanh: str                       # inline | job | run_con
    job_type: str = ""
    app_name: str = ""
    side_effect: str = "none"
    idempotency_key_fields: tuple = ()
    resource: dict = field(default_factory=dict)
    timeout_expr: str = ""
    retry_mac_dinh: dict = field(
        default_factory=lambda: {"max": 0, "backoff_s": 30})
    gia: Gia = field(default_factory=Gia.mien_phi)
    credential: tuple = ()
    quyen: frozenset = frozenset()
    mock: Optional[Callable[[dict], dict]] = None
    cho: tuple = ()

    #: Plan 02 chỉ có `app.*`; node lõi bật cờ này ở plan 04.

    def __post_init__(self) -> None:
        if self.kind not in KIND:
            raise LoiNut(f"kind la: {self.kind!r} (chi {list(KIND)})")
        if self.category not in CATEGORY:
            raise LoiNut(f"category la: {self.category!r}")
        if self.thi_hanh not in ("inline", "job", "run_con"):

    return cau[:60] if cau else ten.replace(".", " ").replace("_", " ").title()
```

```python
        cho=("JOB",), ma_loi=MA_LOI_APP,
        publish_duoc=False)


def chua_du_hop_dong(cong_cu: dict) -> dict:
    """`{tên tool: [trường còn thiếu]}` — spec §4.5 *"khai đủ side_effect,
    idempotency, resource"*.

    ⚠️ `side_effect` **không đo được ở đây**: nó có mặc định `"external"` trong
       `ToolSpec`, nên một tool khai tường minh `"external"` và một tool quên
       khai trông giống hệt nhau. Phép đếm này chỉ nói về `idempotency` và
       `resource` — xem mục "Spec lệch mã thật" của plan 02.
    """
    ra: dict[str, list] = {}
    for ten, cc in (cong_cu or {}).items():
        thieu = []
        if not cc.idempotency_fields:
            thieu.append("idempotency")
        if not cc.resource:
            thieu.append("resource")
        if thieu:
            ra[ten] = thieu
    return ra


def tat_ca(*, cong_cu: Optional[dict] = None,
           nguon_piece: Optional[Callable[[], list]] = None) -> dict:
    """Hợp ba nguồn. Trùng `type` ⇒ `LoiNut` (spec §4.5)."""
    from platform_core.workflow import cong_cu as _nguon
    from platform_core.workflow import nut_loi

    ra: dict[str, NodeDefinition] = dict(nut_loi.tat_ca())
    cc_all = _nguon.tat_ca() if cong_cu is None else cong_cu
    chua_du = chua_du_hop_dong(cc_all)
    for ten, cc in cc_all.items():
        if not cc.workflow or ten in chua_du:
            continue
        nd = tu_toolspec(cc)
        if nd.type in ra:
            raise LoiNut(f"trung type giua hai nguon: {nd.type!r}")
        ra[nd.type] = nd
    for muc in (nguon_piece or NGUON_PIECE)():
        nd = tu_piece(muc)
        if nd.type in ra:
            raise LoiNut(f"trung type giua hai nguon: {nd.type!r}")
        ra[nd.type] = nd
    return ra
```

```python
def type_version_thieu(db, *, dinh_nghia: Optional[dict] = None) -> list:
    """Node của version ĐÃ PUBLISH trỏ tới `type@version` registry không còn.

    ⭐ Đọc bảng chỉ mục `workflow_version_node` (dẫn xuất của `graph`, sinh ở
       `dinh_nghia._sinh_chi_muc`) chứ không duyệt JSON: một truy vấn SQL cho
       toàn hệ thống, và nó chạy được trên PostgreSQL production.
    """
    from sqlalchemy import select

    from platform_core.models import WorkflowVersion, WorkflowVersionNode

    co = dinh_nghia if dinh_nghia is not None else tat_ca()
    ban_do = {t: nd.type_version for t, nd in co.items()}
    hang = db.execute(
        select(WorkflowVersionNode.version_id, WorkflowVersionNode.node_id,
               WorkflowVersionNode.node_type, WorkflowVersionNode.type_version)
        .join(WorkflowVersion, WorkflowVersion.id ==
              WorkflowVersionNode.version_id)
        .where(WorkflowVersion.status == "PUBLISHED")
        .order_by(WorkflowVersionNode.version_id,
                  WorkflowVersionNode.node_id)).all()
    return [{"version_id": v, "node_id": n, "node_type": t, "type_version": tv}
            for v, n, t, tv in hang if ban_do.get(t) != tv]
```

```python
    """Tên backend THẬT đang phục vụ `key` — để `ghi_danh` đặt đúng tiền tố id.

def luu_json(db, *, owner_id: str, key: str, du_lieu: dict,
             snapshot_id: str = "", ttl_class: str = "RUN") -> dict:

    from platform_core import storage as _st

    tho = json.dumps(du_lieu, default=str, ensure_ascii=False).encode("utf-8")

                 kind="json", ttl_class=ttl_class, mime="application/json",
                 size_bytes=len(tho),
                 checksum_sha256=hashlib.sha256(tho).hexdigest(),
                 created_by_snapshot_id=snapshot_id)
    return {"artifact_ref": a.id, "mime": "application/json",
```

```ts
                           hau_to_tap_hop: string; cap_noi_duoc: [string, string][] }

export function noiDuoc(luat: LuatNoiDay, kieuRa: string, kieuVao: string,
                        soCanhDangCo: number, nhanNhieuCanh: boolean): boolean {
  const hop = luat.cap_noi_duoc.some(([a, b]) => a === kieuRa && b === kieuVao)
  if (!hop) return false
  return nhanNhieuCanh || soCanhDangCo < 1
}

export function timNut(nhom: NhomPalette[], type: string): NodeDefinition|undefined {
  for (const n of nhom) {
    const found = n.nut.find(x => x.type === type)
    if (found) return found
  }
  return undefined
}

export function locPalette(nhom: NhomPalette[], truyVan: string): NhomPalette[] {
  const q = truyVan.trim().toLocaleLowerCase('vi')
  if (!q) return nhom
  return nhom
    .map(n => ({ category: n.category, nut: n.nut.filter(x =>
      [x.title, x.description, x.type, x.app_name]
        .some(s => s.toLocaleLowerCase('vi').includes(q))) }))
    .filter(n => n.nut.length > 0)
}
```

```tsx
// Form sinh TU LUOC DO (spec §13.3). ⛔ Khong mot bang `fieldLabels` nao o
// day: nhan la ten tham so cua app, va mo ta den tu `description` cua luoc do.
// Mot bang nhan thu hai la thu `test_967` dang dem hang chet.
import { useState } from 'react'
import type { LuocDo, TruongLuocDo } from '../nodeDefinition'

type Props = {
  luocDo: LuocDo
  giaTri: Record<string, unknown>
  readOnly?: boolean
  onChange: (ten: string, gia_tri: unknown) => void
}

const laBieuThuc = (v: unknown) => typeof v === 'string' && v.includes('{{')

function OJson({ ten, gia, readOnly, onChange }: {
  ten: string; gia: unknown; readOnly?: boolean
  onChange: (v: unknown) => void }) {
  const [tho, setTho] = useState(() =>
    gia === undefined ? '' : typeof gia === 'string' ? gia : JSON.stringify(gia))
  const [loi, setLoi] = useState(false)
  return <>
    <textarea id={ten} disabled={readOnly} value={tho} rows={4}
      onChange={e => {
        const v = e.target.value
        setTho(v)
        if (laBieuThuc(v)) { setLoi(false); onChange(v); return }
        try { onChange(JSON.parse(v)); setLoi(false) } catch { setLoi(true) }
      }} />
    {loi && <small className="o-loi">JSON không hợp lệ</small>}
  </>
}

function MotO({ ten, luat, bat_buoc, gia, readOnly, onChange }: {
  ten: string; luat: TruongLuocDo; bat_buoc: boolean; gia: unknown
  readOnly?: boolean; onChange: (v: unknown) => void }) {
  const khoa = readOnly || luat.readOnly === true
  const nhan = `${ten}${bat_buoc ? ' *' : ''}`

  if (luat.enum?.length) {
    const soHoc = luat.type === 'integer' || luat.type === 'number'
    return <label htmlFor={ten}>{nhan}
      <select id={ten} disabled={khoa} value={String(gia ?? luat.default ?? '')}
        onChange={e => onChange(soHoc ? Number(e.target.value) : e.target.value)}>
        <option value="">— chọn —</option>
        {luat.enum.map(v => <option key={String(v)} value={String(v)}>
          {luat['x-ltt-nhan']?.[String(v)] ?? String(v)}</option>)}
      </select>
      {luat.description && <small>{luat.description}</small>}
    </label>
  }

  if (luat.type === 'boolean') {
    return <label htmlFor={ten} className="o-check">
      <input id={ten} type="checkbox" disabled={khoa}
        checked={Boolean(gia ?? luat.default ?? false)}
        onChange={e => onChange(e.target.checked)} />{nhan}</label>
  }

  if (luat.type === 'object' || luat.type === 'array') {
    return <label htmlFor={ten}>{nhan}
      <OJson ten={ten} gia={gia ?? luat.default} readOnly={khoa}
        onChange={onChange} /></label>
  }

  if (luat['x-ltt-control'] === 'textarea') {
    return <label htmlFor={ten}>{nhan}
      <textarea id={ten} disabled={khoa} rows={3}
        value={String(gia ?? luat.default ?? '')}
        onChange={e => onChange(e.target.value)} /></label>
  }

  const so = luat.type === 'integer' || luat.type === 'number'
  return <label htmlFor={ten}>{nhan}
    <input id={ten} disabled={khoa}
      type={so && !laBieuThuc(gia) ? 'number' : 'text'}
      min={luat.minimum} max={luat.maximum} maxLength={luat.maxLength}
      value={String(gia ?? luat.default ?? '')}
      onChange={e => {
        const v = e.target.value
        // ⭐ Mot o SO van phai nhan duoc `{{ ... }}` (spec §4.6: moi gia tri
        //    co the la literal HOAC bieu thuc). Ep `Number()` o day bien mot
        //    tham chieu hop le thanh `NaN` — im lang.
        if (!so || laBieuThuc(v)) { onChange(v); return }
        onChange(luat.type === 'integer' ? Number.parseInt(v, 10) : Number(v))
      }} />
    {luat.description && <small>{luat.description}</small>}
  </label>
}

export function FormTuLuocDo({ luocDo, giaTri, readOnly, onChange }: Props) {
  const muc = Object.entries(luocDo.properties ?? {})
  if (!muc.length) return <p className="form-trong">Node này không có tham số.</p>
  return <div className="form-luoc-do">
    {muc.map(([ten, luat]) => <MotO key={ten} ten={ten} luat={luat}
      bat_buoc={(luocDo.required ?? []).includes(ten)} gia={giaTri[ten]}
      readOnly={readOnly} onChange={v => onChange(ten, v)} />)}
  </div>
}
```

## Spec lệch mã thật — ba chỗ, kiểm bằng cách đọc mã ngày 23/09

**1. `ToolSpec.params_schema` KHÔNG phải JSON Schema.** Spec §4.5 viết *"rỗng ⇒ adapter suy: mỗi thuộc tính `params_schema` là cổng vào kiểu suy từ JSON Schema"*. Thực tế `registry.TOOL_PARAM_TYPES` là một từ vựng riêng (`string int number bool enum model_ref text image image[] video video[] audio json json[]`) với các khoá `values`/`min`/`max`/`max_length`/`source`/`depends_on`/`measured_from`. Không có bước dịch thì `NodeDefinition.tham_so` của §4.6 không tồn tại được. Plan 02 giải bằng `luoc_do.py` (Task 4) và spec nên sửa câu đó thành *"suy từ từ vựng LTT qua `workflow/luoc_do.py`"*. Ledger 22/09 đã ghi nửa đầu của lệch này.

**2. "Khai đủ `side_effect`" không đo được.** Spec §4.5: *"Tool chỉ lên palette workflow khi khai đủ `side_effect`, `idempotency`, `resource` (hôm nay 4/19)"*. Nhưng `ToolSpec.side_effect` có **mặc định `"external"`**, nên một tool khai tường minh và một tool quên khai là **cùng một giá trị** — không phép kiểm nào phân biệt được. Con số 4/19 trong spec thực ra đo `idempotency` + `resource` (đúng 4: `audio_pause.run`, `prompt_canh.run`, `transcript.run`, `video_assembly.run`). Plan 02 đo đúng hai trường ấy (`nut.chua_du_hop_dong`) và nói thẳng ra trong docstring. Muốn đo cả `side_effect` thì phải đổi mặc định thành một sentinel (`""`) — một thay đổi phá vỡ với mọi tool, nên **không** làm trong plan 02.

**3. `types_compatible()` / `TYPE_MIGRATIONS` của Langflow (§4.7).** Spec bảo mượn *"~15 dòng, MIT"*. Tôi **không xác minh được** hai định danh này tồn tại trong Langflow ở phiên bản nào, và luật *"dùng lại trước, tự viết sau"* đòi chứng minh trước khi chép. Ngoài ra `he_kieu.ep_duoc` hiện chỉ có **một** luật (`T → T[]`) có chủ ý, kèm lý do viết sẵn trong mã (*"một phép biến đổi ngầm không hiện trên đồ thị"*) — thêm một bảng đổi tên kiểu vào đó là nới lỏng đúng thứ nó cố giữ chặt. Plan 02 **hoãn**: khi nào có một kiểu thật bị đổi tên (plan 08 với piece) thì tra Langflow qua GitHub MCP rồi quyết, và ghi notice vào `THIRD_PARTY_NOTICES.md` nếu chép.
