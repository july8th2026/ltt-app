import json, os, time, uuid, urllib.request

BASE = "https://app.lttstudios.com/api/v1"

def goi(method, path, body=None, them=None):
    data = json.dumps(body).encode("utf-8") if body is not None else None
    headers = {
        "Authorization": "Bearer " + os.environ["LTT_API_KEY"],
        "Content-Type": "application/json",
        # BAT BUOC. Chuoi urllib tu dien ("Python-urllib/3.x") bi
        # Cloudflare chan bang 403 truoc khi toi may chu — trong y
        # het khoa sai. Dat bat cu User-Agent nao khac la qua.
        "User-Agent": "ltt-client/1.0",
    }
    headers.update(them or {})
    req = urllib.request.Request(BASE + path, data=data,
                                 headers=headers, method=method)
    return json.load(urllib.request.urlopen(req))

vao = {"text": "{{ vi_du }}", "preset": "{{ mac_dinh }}"}
bao_gia = goi("POST", "/quotes", {"tool": "{{ ten_tool }}", "input": vao})
job = goi("POST", "/jobs",
          {"tool": "{{ ten_tool }}", "quote": bao_gia["quote"], "input": vao},
          {"Idempotency-Key": uuid.uuid4().hex})
while job["status"] in ("queued", "running"):
    time.sleep(2)
    job = goi("GET", "/jobs/" + job["id"])
print(job)

// Chay o PHIA MAY CHU (Node 18+). Dung dat khoa trong ma trinh duyet.
const BASE = "https://app.lttstudios.com/api/v1";
const goi = async (method, path, body, them = {}) => {
  const r = await fetch(BASE + path, {
    method,
    headers: {
      "Authorization": `Bearer ${process.env.LTT_API_KEY}`,
      "Content-Type": "application/json",
      ...them,
    },
    body: body ? JSON.stringify(body) : undefined,
  });
  return r.json();
};
